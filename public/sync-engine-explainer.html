V Vue Composables

The sync engine is initialized at the app root and provides context to all components via Vue's injection pattern.

// app/composables/useIDBSyncEngine.ts
export const [useIDBSyncEngineProvider, useIDBSyncEngine] =
  createInjectionState((options: { peer?: string }) => {
    const { db, isReady, init } = useIndexedDB()
    const deviceId = useDeviceId()
    const { status: wsStatus, send, data } = useWebSocket(peer, {
      autoReconnect: {
        retries: 5,
        delay: (retries) => Math.min(1000 * 2 ** (retries - 1), 30000)
      }
    })

    return { db, deviceId, wsStatus, send, ... }
  })

CRUD operations use useIDBCoList which handles IndexedDB transactions and sync tracking:

// app/composables/useIDBCoList.ts
async function create(data: T): Promise<string> {
  const id = crypto.randomUUID()
  const item: SyncItem = {
    id,
    data,
    createdAt: Date.now(),
    updatedAt: Date.now(),
    deviceId: engine.deviceId,
    deleted: false
  }

  await txQueue.enqueue(db, collection, 'readwrite', (tx) => {
    tx.objectStore(collection).put(item)
  })

  markUnsynced(id)
  broadcast({ type: 'entity_changed', collection, entityId: id })
  return id
}

Transaction queue serializes operations to prevent conflicts:

// app/utils/idb-transaction-queue.ts
class TransactionQueue {
  private queue: Array<() => Promise<void>> = []
  private processing = false

  async enqueue<T>(
    db: IDBDatabase,
    stores: string | string[],
    mode: IDBTransactionMode,
    operation: (tx: IDBTransaction) => Promise<T>
  ): Promise<T> {
    return new Promise((resolve, reject) => {
      this.queue.push(async () => {
        const tx = db.transaction(stores, mode)
        const result = await operation(tx)
        resolve(result)
      })
      this.processQueue()
    })
  }
}

W WebSocket Sync

Server-side WebSocket handler using Nitro's CrossWS support:

// server/routes/_ws.ts
export default defineWebSocketHandler({
  open(peer) {
    peer.subscribe('sync-channel')
    peer.send(JSON.stringify({
      type: 'welcome',
      peerId: peer.id
    }))
  },

  message(peer, message) {
    const data = JSON.parse(message.text())

    switch (data.type) {
      case 'sync':
        // Broadcast changes to all peers
        peer.publish('sync-channel', JSON.stringify(data))
        break
      case 'ping':
        peer.send(JSON.stringify({ type: 'pong' }))
        break
    }
  },

  close(peer) {
    console.log('Client disconnected:', peer.id)
  }
})

T Cross-Tab Sync

BroadcastChannel enables instant sync between browser tabs without server roundtrip:

// app/composables/useCrossTabSync.ts
interface CrossTabChange {
  type: 'entity_changed'
  collection: string      // e.g., 'todos'
  entityId: string
  operation: 'create' | 'update' | 'delete'
  sourceTabId: string     // Filter out own messages
  timestamp: number
}

export function useCrossTabSync(tabId: string) {
  const channel = new BroadcastChannel('sync-engine-cross-tab')

  function broadcast(change: Omit<CrossTabChange, 'sourceTabId' | 'timestamp'>) {
    channel.postMessage({
      ...change,
      sourceTabId: tabId,
      timestamp: Date.now()
    })
  }

  channel.onmessage = (event) => {
    if (event.data.sourceTabId === tabId) return // Ignore own
    // Handle incoming change...
  }

  return { broadcast, onMessage: channel.onmessage }
}

C Conflict Resolution (LWW)

Last-Write-Wins with deterministic device ID tiebreaker ensures all replicas converge to the same value:

// utils/sync-protocol.ts
export function shouldIncomingWin(
  incoming: Todo,
  existing: Todo
): boolean {
  // 1. Higher timestamp wins
  if (incoming.updated_at > existing.updated_at) {
    return true
  }

  // 2. Equal timestamps: deterministic tiebreaker
  if (incoming.updated_at === existing.updated_at) {
    // Lexicographic comparison ensures same result everywhere
    return incoming.device_id > existing.device_id
  }

  return false
}

Server-side conflict detection in batchUpsert:

// server/utils/db.ts
function batchUpsert(tableName, items) {
  const conflicts = []

  for (const item of items) {
    const existing = getItemById(tableName, item.id)

    if (existing && existing.updatedAt >= item.updatedAt) {
      // Server has newer or equal version
      conflicts.push(existing)
    } else {
      // Insert or update
      db.prepare(`INSERT OR REPLACE INTO ...`).run(item)
    }
  }

  return { stored: items.length - conflicts.length, conflicts }
}

S Server Database

SQLite with better-sqlite3 provides persistent server-side storage:

// server/utils/db.ts
function getDatabase() {
  const db = new Database('./data/sync.db')

  // Performance optimizations
  db.pragma('journal_mode = WAL')
  db.pragma('synchronous = NORMAL')
  db.pragma('foreign_keys = ON')

  // Create tables with sync metadata
  db.exec(`
    CREATE TABLE IF NOT EXISTS todos (
      id TEXT PRIMARY KEY,
      text TEXT NOT NULL,
      completed INTEGER DEFAULT 0,
      created_at INTEGER NOT NULL,
      updated_at INTEGER NOT NULL,
      device_id TEXT NOT NULL,
      deleted INTEGER DEFAULT 0
    );
    CREATE INDEX IF NOT EXISTS idx_todos_updated_at
      ON todos(updated_at);
  `)

  return db
}
